app = FastAPI(
    title=settings.PROJECT_NAME,
    version=settings.APP_VERSION,
    openapi_url=f"{settings.API_V1_STR}/openapi.json",
)

# CORS Middleware
if settings.BACKEND_CORS_ORIGINS:
    app.add_middleware(
        CORSMiddleware,
        allow_origins=[str(origin) for origin in settings.BACKEND_CORS_ORIGINS],
        allow_credentials=True,
        allow_methods=["GET", "POST", "PUT", "DELETE", "OPTIONS"],
        allow_headers=["*"],
    )

# Rate Limiting Middleware
from autosoc_api.middleware.rate_limit import RateLimitMiddleware
app.add_middleware(RateLimitMiddleware, rate_limit=50, window=60)

# Request ID Middleware
@app.middleware("http")
async def add_request_id_and_log(request: Request, call_next: Callable[[Request], Awaitable[Response]]) -> Response:
    request_id = request.headers.get("X-Request-ID")
    if not request_id or len(request_id) > 50 or not request_id.isalnum():
        request_id = str(uuid.uuid4())
    
    # Store request_id in request state
    request.state.request_id = request_id
    
    start_time = time.time()
    
    try:
        response = await call_next(request)
    except Exception as exc:
        logger.error(f"Unhandled Exception: {exc}", extra={"request_id": request_id})
        # Note: Do not return stack trace to client
        response = JSONResponse(
            status_code=500,
            content={
                "error_code": "INTERNAL_SERVER_ERROR",
                "message": "An unexpected error occurred.",
                "request_id": request_id,
            }
        )
        
    process_time = time.time() - start_time
    response.headers["X-Request-ID"] = request_id
    response.headers["X-Process-Time"] = str(process_time)
    
    return response

# Validation Error Handler
@app.exception_handler(ValidationError)
async def validation_exception_handler(request: Request, exc: ValidationError) -> JSONResponse:
    request_id = getattr(request.state, "request_id", "unknown")
    return JSONResponse(
        status_code=422,
        content={
            "error_code": "VALIDATION_ERROR",
            "message": "Input validation failed.",
            "details": exc.errors(),
            "request_id": request_id,
        },
    )

app.include_router(api_router, prefix=settings.API_V1_STR)

@app.get("/")
def root() -> dict[str, str]:
    return {
        "service": settings.PROJECT_NAME,
        "api_version": settings.API_VERSION,
        "environment": settings.ENVIRONMENT,
        "phase": "Phase 5 - Core Backend",
        "docs_url": "/docs"
    }

@router.post("/{incident_id}/agents/investigate", response_model=AgentExecutionResponse)
async def investigate_incident(
    incident_id: uuid.UUID,
    background_tasks: BackgroundTasks,
    request: AgentExecutionCreate = None,
    db: AsyncSession = Depends(get_db_session),
    # Only SOC Analysts and Admins can start an investigation
    current_user: Principal = Depends(require_roles(["soc_analyst", "admin"]))
) -> Any:
    """
    Start an Agentic AI investigation for an incident.
    """
    incident = await db.get(Incident, incident_id)
    if not incident:
        raise HTTPException(status_code=404, detail="Incident not found")
        
    if not request:
        request = AgentExecutionCreate(incident_id=incident_id)

    # Check for existing running execution
    result = await db.execute(
        select(AgentExecution)
        .where(AgentExecution.incident_id == incident_id)
        .where(AgentExecution.status.in_([AgentExecutionStatus.QUEUED, AgentExecutionStatus.RUNNING]))
    )
    existing = result.scalars().first()
    if existing:
        raise HTTPException(
            status_code=409, 
            detail=f"An investigation is already running for this incident (execution_id: {existing.id})"
        )

    execution = AgentExecution(
        incident_id=incident_id,
        workflow_type=request.workflow_type,
        status=AgentExecutionStatus.QUEUED,
        requested_by=current_user.username
    )
    
    db.add(execution)
    await db.commit()
    await db.refresh(execution)
    
    # Trigger background execution (via Redpanda/HTTP)
    background_tasks.add_task(trigger_agent_worker, str(execution.id), str(incident_id), request.workflow_type)
    
    return execution

@router.post("/agent-executions/{execution_id}/cancel", status_code=202)
async def cancel_agent_execution(
    execution_id: uuid.UUID,
    db: AsyncSession = Depends(get_db_session),
    current_user: Principal = Depends(require_roles(["soc_analyst", "admin"]))
) -> Any:
    """
    Cancel a running agent execution safely.
    """
    execution = await db.get(AgentExecution, execution_id)
    if not execution:
        raise HTTPException(status_code=404, detail="Execution not found")
        
    if execution.status not in [AgentExecutionStatus.QUEUED, AgentExecutionStatus.RUNNING]:
        raise HTTPException(status_code=400, detail=f"Cannot cancel execution in state {execution.status}")
        
    execution.status = AgentExecutionStatus.CANCELLED
    await db.commit()
    
    return {"message": "Cancellation requested successfully"}

    )
    
    return alert

import httpx
from typing import Optional, Dict, Any, List
from fastapi import APIRouter, Depends, HTTPException, Query
from autosoc_api.core.config import settings

router = APIRouter()

async def get_opensearch_client() -> httpx.AsyncClient:
    client = httpx.AsyncClient(base_url=settings.OPENSEARCH_HOSTS)
    try:
        yield client
    finally:
        await client.aclose()

@router.get("/dashboard/summary", response_model=Dict[str, Any])
async def get_dashboard_summary(
    client: httpx.AsyncClient = Depends(get_opensearch_client)
) -> Dict[str, Any]:
    """
    Get aggregated dashboard summary (events by category, timeline).
    """
    query = {
        "size": 0,
        "query": {
            "range": {
                "event_time": {
                    "gte": "now-24h/h",
                    "lte": "now/h"
                }
            }
        },
        "aggs": {
            "categories": {
                "terms": {"field": "event_category", "size": 10}
            },
            "timeline": {
                "date_histogram": {
                    "field": "event_time",
                    "fixed_interval": "1h"
                }
            },
            "sources": {
                "terms": {"field": "source_type", "size": 10}
            }
        }
    }
    
    try:
        resp = await client.post("/autosoc-events-v1-*/_search", json=query)
        resp.raise_for_status()
        data = resp.json()
        
        aggs = data.get("aggregations", {})
        
        return {
            "total_events_24h": data.get("hits", {}).get("total", {}).get("value", 0),
            "categories": [{"category": b["key"], "count": b["doc_count"]} for b in aggs.get("categories", {}).get("buckets", [])],
            "timeline": [{"time": b["key_as_string"], "count": b["doc_count"]} for b in aggs.get("timeline", {}).get("buckets", [])],
            "sources": [{"source": b["key"], "count": b["doc_count"]} for b in aggs.get("sources", {}).get("buckets", [])]
        }
    except httpx.HTTPStatusError as e:
        raise HTTPException(status_code=502, detail=f"OpenSearch upstream error: {e}")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/", response_model=Dict[str, Any])
async def search_events(
    category: Optional[str] = Query(None, description="Filter by event category"),
    source_ip: Optional[str] = Query(None, description="Filter by source IP"),
    limit: int = Query(50, ge=1, le=1000),
    client: httpx.AsyncClient = Depends(get_opensearch_client)
) -> Dict[str, Any]:
    """
    Query normalized/enriched security events safely without allowing raw Lucene DSL.
    """
    # Build safe OpenSearch query
    must_clauses: List[Dict[str, Any]] = []
    
    if category:
        must_clauses.append({"term": {"event_category": category}})
    if source_ip:
        must_clauses.append({"term": {"source_ip": source_ip}})
        
    query = {
        "size": limit,
        "sort": [{"event_time": {"order": "desc"}}],
        "query": {
            "bool": {
                "must": must_clauses
            }
        } if must_clauses else {"match_all": {}}
    }
    
    try:
        resp = await client.post("/autosoc-events-v1-*/_search", json=query)
        resp.raise_for_status()
        data = resp.json()
        
        hits = data.get("hits", {}).get("hits", [])
        events = [hit["_source"] for hit in hits]
        
        return {
            "total": data.get("hits", {}).get("total", {}).get("value", 0),
            "events": events
        }
    except httpx.HTTPStatusError as e:
        raise HTTPException(status_code=502, detail=f"OpenSearch upstream error: {e}")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/{event_id}", response_model=Dict[str, Any])
async def get_event(
    event_id: str,
    client: httpx.AsyncClient = Depends(get_opensearch_client)
) -> Dict[str, Any]:

    """

                "normalized_event_id": event_id
            }
        }
    }

        hits = data.get("hits", {}).get("hits", [])
        if not hits:
            raise HTTPException(status_code=404, detail="Event not found")
            
        return hits[0]["_source"]
    except httpx.HTTPStatusError as e:
        raise HTTPException(status_code=502, detail=f"OpenSearch upstream error: {e}")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

        return existing

        playbook_id=recommendation.playbook_id,

        idempotency_key=idempotency_key
    )

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from autosoc_api.auth.models import Principal

# This schema will be used in OpenAPI for Bearer token documentation
oauth2_scheme = HTTPBearer(auto_error=False)

def get_current_principal(
    token: HTTPAuthorizationCredentials = Depends(oauth2_scheme)
) -> Principal:
    """
    Dependency that validates the Bearer token and returns the current Principal.
    Rejects missing or invalid tokens with HTTP 401.
    """
    if not token:
        # In a test environment we might bypass this if explicitly configured, 
        # but for Phase 5 we strictly enforce it unless using dependency overrides in pytest.
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not authenticated",
            headers={"WWW-Authenticate": "Bearer"},
        )
    
    principal = validator.validate_token(token.credentials)
    principal.permissions = get_permissions_for_roles(principal.roles)
    return principal

def require_permission(required_permission: str):
    """
    Dependency generator that checks if the Principal has the required permission.
    Returns HTTP 403 if unauthorized.
    """
    def permission_checker(principal: Principal = Depends(get_current_principal)) -> Principal:
        if required_permission not in principal.permissions:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Not enough permissions"
            )
        return principal
    return permission_checker

def require_roles(allowed_roles: list[str]):
    """
    Dependency generator that checks if the Principal has at least one of the allowed roles.
    Returns HTTP 403 if unauthorized.
    """
    def role_checker(principal: Principal = Depends(get_current_principal)) -> Principal:
        if not set(allowed_roles).intersection(set(principal.roles)):
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Not enough roles"
            )
        return principal
    return role_checker

import jwt

from typing import Dict, Any, Optional

logger = get_logger(__name__)

class KeycloakJWTValidator:
    def __init__(self):
        self.jwks_url = settings.KEYCLOAK_JWKS_URL
        self.client_id = settings.KEYCLOAK_CLIENT_ID
        self.issuer = f"{settings.KEYCLOAK_SERVER_URL}/realms/{settings.KEYCLOAK_REALM}"
        self._jwks_client = jwt.PyJWKClient(self.jwks_url)

    def validate_token(self, token: str) -> Principal:
        try:
            signing_key = self._jwks_client.get_signing_key_from_jwt(token)
            
            # Decode and validate signature, expiry, issuer, and audience
            payload = jwt.decode(
                token,
                signing_key.key,
                algorithms=["RS256"],
                issuer=self.issuer,
                audience="account", # Default Keycloak audience if none mapped, or we can check azp
                options={"verify_aud": False} # We check azp instead below
            )
            
            # Keycloak uses 'azp' (Authorized Party) instead of 'aud' often for OIDC
            if "azp" in payload and payload["azp"] != self.client_id:
                # We log but allow for dev if needed, or enforce:
                pass # Enforce audience if strictly required by policy
                
            realm_access = payload.get("realm_access", {})
            roles = realm_access.get("roles", [])
            
            return Principal(
                subject_id=payload.get("sub", ""),
                username=payload.get("preferred_username", ""),
                email=payload.get("email"),
                roles=roles,
                issuer=self.issuer
            )
            
        except jwt.ExpiredSignatureError:
            logger.warning("Token expired")
            raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Token expired")
        except jwt.InvalidIssuerError:
            logger.warning("Invalid token issuer")
            raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid issuer")
        except jwt.PyJWKClientError as e:
            logger.error(f"JWKS retrieval failed: {e}")
            raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Could not validate credentials")
        except Exception as e:
            logger.warning(f"Token validation failed: {e}")
            raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid token")

validator = KeycloakJWTValidator()

import logging
import sys

def setup_logging(log_level: str = "INFO") -> None:
    """Configures structured logging foundation."""
    
    logger = logging.getLogger("autosoc_api")
    
    # Do not reconfigure if already configured
    if logger.handlers:
        return
        
    logger.setLevel(getattr(logging, log_level.upper(), logging.INFO))
    
    handler = logging.StreamHandler(sys.stdout)
    handler.setLevel(getattr(logging, log_level.upper(), logging.INFO))
    
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    )
    handler.setFormatter(formatter)

engine = create_async_engine(
    str(settings.SQLALCHEMY_DATABASE_URI),
    echo=settings.DEBUG,
    pool_size=settings.DB_POOL_SIZE,
    max_overflow=settings.DB_MAX_OVERFLOW,
    pool_timeout=settings.DB_POOL_TIMEOUT,
    pool_recycle=settings.DB_POOL_RECYCLE,
)

# Create Async Session Maker
AsyncSessionLocal = async_sessionmaker(
    engine,
    class_=AsyncSession,
    expire_on_commit=False,
    autocommit=False,
    autoflush=False,
)

async def get_db_session() -> AsyncGenerator[AsyncSession, None]:
    """
    FastAPI Dependency to yield a database session.
    Ensures safe session closure on request completion.
    """
    async with AsyncSessionLocal() as session:
        try:
            yield session
        finally:
            await session.close()

from typing import Generic, TypeVar, Type, Optional, Sequence
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, asc, desc
from autosoc_api.db.base import Base

ModelType = TypeVar("ModelType", bound=Base)

class BaseRepository(Generic[ModelType]):
    def __init__(self, model: Type[ModelType]):
        self.model = model

    async def get(self, db: AsyncSession, id: str) -> Optional[ModelType]:
        result = await db.execute(select(self.model).where(self.model.id == id))
        return result.scalars().first()

    async def get_multi(
        self, db: AsyncSession, *, skip: int = 0, limit: int = 100, sort_by: str = "created_at", sort_desc: bool = True
    ) -> Sequence[ModelType]:
        # Validate sort column to prevent SQL injection
        sort_column = getattr(self.model, sort_by, getattr(self.model, "created_at"))
        
        stmt = select(self.model).offset(skip).limit(limit)
        if sort_desc:
            stmt = stmt.order_by(desc(sort_column))
        else:
            stmt = stmt.order_by(asc(sort_column))
            
        result = await db.execute(stmt)
        return result.scalars().all()

    async def create(self, db: AsyncSession, *, obj_in: dict) -> ModelType:
        db_obj = self.model(**obj_in)
        db.add(db_obj)
        await db.commit()
        await db.refresh(db_obj)
        return db_obj

    async def update(self, db: AsyncSession, *, db_obj: ModelType, obj_in: dict) -> ModelType:
        for field, value in obj_in.items():
            setattr(db_obj, field, value)
        db.add(db_obj)
        await db.commit()
        await db.refresh(db_obj)
        return db_obj

    async def delete(self, db: AsyncSession, *, id: str) -> Optional[ModelType]:
        obj = await self.get(db, id)
        if obj:
            await db.delete(obj)
            await db.commit()
        return obj

import time
import asyncio
from collections import defaultdict
from fastapi import Request, HTTPException
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.responses import JSONResponse
import logging

logger = logging.getLogger(__name__)

# Basic in-memory rate limiter for academic/dev demonstration
# In production, use Redis.
class RateLimitMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, rate_limit: int = 100, window: int = 60):
        super().__init__(app)
        self.rate_limit = rate_limit
        self.window = window
        self.requests = defaultdict(list)

    async def dispatch(self, request: Request, call_next):
        # Only rate limit specific sensitive paths for SOC
        sensitive_paths = ["/responses", "/agents", "/incidents"]
        
        path = request.url.path
        if any(sp in path for sp in sensitive_paths) and request.method in ["POST", "PUT", "DELETE"]:
            client_ip = request.client.host
            now = time.time()
            
            # Clean up old requests
            self.requests[client_ip] = [req_time for req_time in self.requests[client_ip] if now - req_time < self.window]
            
            if len(self.requests[client_ip]) >= self.rate_limit:
                logger.warning(f"Rate limit exceeded for IP {client_ip} on path {path}")
                return JSONResponse(
                    status_code=429,
                    content={"error_code": "RATE_LIMIT_EXCEEDED", "message": "Too many requests. Please try again later."}
                )
            
            self.requests[client_ip].append(now)

        return await call_next(request)

class IncidentService:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def get_all(self, skip: int = 0, limit: int = 100) -> List[Incident]:
        result = await self.db.execute(select(Incident).offset(skip).limit(limit))
        return list(result.scalars().all())

    async def get_by_id(self, incident_id: uuid.UUID) -> Incident:
        result = await self.db.execute(select(Incident).where(Incident.id == incident_id))
        incident = result.scalar_one_or_none()
        if not incident:
            raise HTTPException(status_code=404, detail="Incident not found")
        return incident

    async def create_note(self, incident_id: uuid.UUID, note_in: IncidentNoteCreate, author_id: str) -> IncidentNote:
        # Verify incident exists
        await self.get_by_id(incident_id)
        
        note = IncidentNote(

        )

import uuid
from typing import Optional
from sqlalchemy.ext.asyncio import AsyncSession
from fastapi import HTTPException
from autosoc_api.db.models.incident import Incident, IncidentStatus
from autosoc_api.services.incident import IncidentService

class IncidentTransitionService:
    def __init__(self, db: AsyncSession):
        self.db = db
        self.incident_service = IncidentService(db)

    async def update_status(self, incident_id: uuid.UUID, new_status: IncidentStatus, user_id: str, reason: Optional[str] = None) -> Incident:
        incident = await self.incident_service.get_by_id(incident_id)
        
        # Define allowed transitions
        # Simplistic approach for now
        allowed_transitions = {
            IncidentStatus.NEW: [IncidentStatus.TRIAGED, IncidentStatus.FALSE_POSITIVE, IncidentStatus.CLOSED],
            IncidentStatus.TRIAGED: [IncidentStatus.INVESTIGATING, IncidentStatus.FALSE_POSITIVE, IncidentStatus.CLOSED],
            IncidentStatus.INVESTIGATING: [IncidentStatus.AWAITING_APPROVAL, IncidentStatus.CONTAINED, IncidentStatus.RESOLVED],
            IncidentStatus.AWAITING_APPROVAL: [IncidentStatus.CONTAINED, IncidentStatus.RESOLVED, IncidentStatus.INVESTIGATING],
            IncidentStatus.CONTAINED: [IncidentStatus.RESOLVED],
            IncidentStatus.RESOLVED: [IncidentStatus.CLOSED],
            IncidentStatus.FALSE_POSITIVE: [IncidentStatus.NEW],
            IncidentStatus.CLOSED: [IncidentStatus.NEW, IncidentStatus.INVESTIGATING]
        }
        
        if new_status not in allowed_transitions.get(incident.status, []):
            raise HTTPException(status_code=400, detail=f"Invalid transition from {incident.status} to {new_status}")
            
        old_status = incident.status
        incident.status = new_status
        if new_status == IncidentStatus.FALSE_POSITIVE and reason:
            incident.false_positive_reason = reason
        if new_status == IncidentStatus.CLOSED and reason:
            incident.closure_reason = reason
            
        from autosoc_api.db.models.audit_log import AuditLog
        audit = AuditLog(
            actor_id=user_id,
            action="UPDATE_INCIDENT_STATUS",
            target_type="incident",
            target_id=str(incident_id),
            status="success",
            metadata_json={"old_status": old_status.value, "new_status": new_status.value}
        )
        self.db.add(audit)
        
        await self.db.commit()
        await self.db.refresh(incident)
        return incident
