import httpx
from typing import Optional, Dict, Any, List
from fastapi import APIRouter, Depends, HTTPException, Query
from autosoc_api.core.config import settings

router = APIRouter()

async def get_opensearch_client() -> httpx.AsyncClient:
    client = httpx.AsyncClient(base_url=settings.OPENSEARCH_HOSTS)
    try:
        yield client
    finally:
        await client.aclose()

@router.get("/dashboard/summary", response_model=Dict[str, Any])
async def get_dashboard_summary(
    client: httpx.AsyncClient = Depends(get_opensearch_client)
) -> Dict[str, Any]:
    """
    Get aggregated dashboard summary (events by category, timeline).
    """
    query = {
        "size": 0,
        "query": {
            "range": {
                "event_time": {
                    "gte": "now-24h/h",
                    "lte": "now/h"
                }
            }
        },
        "aggs": {
            "categories": {
                "terms": {"field": "event_category", "size": 10}
            },
            "timeline": {
                "date_histogram": {
                    "field": "event_time",
                    "fixed_interval": "1h"
                }
            },
            "sources": {
                "terms": {"field": "source_type", "size": 10}
            }
        }
    }
    
    try:
        resp = await client.post("/autosoc-events-v1-*/_search", json=query)
        resp.raise_for_status()
        data = resp.json()
        
        aggs = data.get("aggregations", {})
        
        return {
            "total_events_24h": data.get("hits", {}).get("total", {}).get("value", 0),
            "categories": [{"category": b["key"], "count": b["doc_count"]} for b in aggs.get("categories", {}).get("buckets", [])],
            "timeline": [{"time": b["key_as_string"], "count": b["doc_count"]} for b in aggs.get("timeline", {}).get("buckets", [])],
            "sources": [{"source": b["key"], "count": b["doc_count"]} for b in aggs.get("sources", {}).get("buckets", [])]
        }
    except httpx.HTTPStatusError as e:
        raise HTTPException(status_code=502, detail=f"OpenSearch upstream error: {e}")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/", response_model=Dict[str, Any])
async def search_events(
    category: Optional[str] = Query(None, description="Filter by event category"),
    source_ip: Optional[str] = Query(None, description="Filter by source IP"),
    limit: int = Query(50, ge=1, le=1000),
    client: httpx.AsyncClient = Depends(get_opensearch_client)
) -> Dict[str, Any]:
    """
    Query normalized/enriched security events safely without allowing raw Lucene DSL.
    """
    # Build safe OpenSearch query
    must_clauses: List[Dict[str, Any]] = []
    
    if category:
        must_clauses.append({"term": {"event_category": category}})
    if source_ip:
        must_clauses.append({"term": {"source_ip": source_ip}})
        
    query = {
        "size": limit,
        "sort": [{"event_time": {"order": "desc"}}],
        "query": {
            "bool": {
                "must": must_clauses
            }
        } if must_clauses else {"match_all": {}}
    }
    
    try:
        resp = await client.post("/autosoc-events-v1-*/_search", json=query)
        resp.raise_for_status()
        data = resp.json()
        
        hits = data.get("hits", {}).get("hits", [])
        events = [hit["_source"] for hit in hits]
        
        return {
            "total": data.get("hits", {}).get("total", {}).get("value", 0),
            "events": events
        }
    except httpx.HTTPStatusError as e:
        raise HTTPException(status_code=502, detail=f"OpenSearch upstream error: {e}")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/{event_id}", response_model=Dict[str, Any])
async def get_event(
    event_id: str,
    client: httpx.AsyncClient = Depends(get_opensearch_client)
) -> Dict[str, Any]:
    """
    Get a single event by ID.
    """
    query = {
        "query": {
            "term": {
                "normalized_event_id": event_id
            }
        }
    }
    try:
        resp = await client.post("/autosoc-events-v1-*/_search", json=query)
        resp.raise_for_status()
        data = resp.json()
        
        hits = data.get("hits", {}).get("hits", [])
        if not hits:
            raise HTTPException(status_code=404, detail="Event not found")
            
        return hits[0]["_source"]
    except httpx.HTTPStatusError as e:
        raise HTTPException(status_code=502, detail=f"OpenSearch upstream error: {e}")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/", response_model=List[AlertResponse])
async def list_alerts(
    db: AsyncSession = Depends(get_db_session),
    user: Principal = Depends(require_roles(["soc_analyst", "admin", "security_manager", "auditor"])),
    status: Optional[str] = None,
    severity: Optional[str] = None,
    limit: int = Query(100, le=1000),
    offset: int = 0
):
    stmt = select(Alert).order_by(desc(Alert.created_at)).limit(limit).offset(offset)
    if status:
        stmt = stmt.where(Alert.status == status)
    if severity:
        stmt = stmt.where(Alert.severity == severity)
        
    result = await db.execute(stmt)
    return result.scalars().all()

    alert = result.scalar_one_or_none()

@router.patch("/{alert_id}/status", response_model=AlertResponse)
async def update_alert_status(
    alert_id: UUID,
    update_data: AlertUpdate,
    db: AsyncSession = Depends(get_db_session),
    user: Principal = Depends(require_roles(["soc_analyst", "admin"]))
):
    stmt = select(Alert).where(Alert.id == alert_id)
    result = await db.execute(stmt)
    alert = result.scalar_one_or_none()
    if not alert:
        raise HTTPException(status_code=404, detail="Alert not found")
        
    old_status = alert.status
    if update_data.status:
        alert.status = update_data.status
        
    await db.commit()
    await db.refresh(alert)
    
    await audit_log(
        db,
        user.subject_id,
        "UPDATE_ALERT_STATUS",
        f"Alert {alert_id} status changed from {old_status} to {alert.status}",
        True
    )
    
    return alert

@router.post("/{incident_id}/agents/investigate", response_model=AgentExecutionResponse)
async def investigate_incident(
    incident_id: uuid.UUID,
    background_tasks: BackgroundTasks,
    request: AgentExecutionCreate = None,
    db: AsyncSession = Depends(get_db_session),
    # Only SOC Analysts and Admins can start an investigation
    current_user: Principal = Depends(require_roles(["soc_analyst", "admin"]))
) -> Any:
    """
    Start an Agentic AI investigation for an incident.
    """
    incident = await db.get(Incident, incident_id)
    if not incident:
        raise HTTPException(status_code=404, detail="Incident not found")
        
    if not request:
        request = AgentExecutionCreate(incident_id=incident_id)

    # Check for existing running execution
    result = await db.execute(
        select(AgentExecution)
        .where(AgentExecution.incident_id == incident_id)
        .where(AgentExecution.status.in_([AgentExecutionStatus.QUEUED, AgentExecutionStatus.RUNNING]))
    )
    existing = result.scalars().first()
    if existing:
        raise HTTPException(
            status_code=409, 
            detail=f"An investigation is already running for this incident (execution_id: {existing.id})"
        )

    execution = AgentExecution(
        incident_id=incident_id,
        workflow_type=request.workflow_type,
        status=AgentExecutionStatus.QUEUED,
        requested_by=current_user.username
    )
    
    db.add(execution)
    await db.commit()
    await db.refresh(execution)
    
    # Trigger background execution (via Redpanda/HTTP)
    background_tasks.add_task(trigger_agent_worker, str(execution.id), str(incident_id), request.workflow_type)
    
    return execution

@router.post("/responses/{recommendation_id}/approve", response_model=ResponseApprovalResponse)
def approve_recommendation(
    recommendation_id: UUID4,
    action: ResponseApprovalAction,
    db: Session = Depends(dependencies.get_db),
    current_user: Principal = Depends(require_permission("responses:approve"))
) -> Any:
    """
    Approve or reject a recommendation.
    """
    recommendation = db.query(ResponseRecommendation).filter(ResponseRecommendation.id == recommendation_id).first()
    if not recommendation:
        raise HTTPException(status_code=404, detail="Recommendation not found")
        
    approval = db.query(ResponseApproval).filter(
        ResponseApproval.recommendation_id == recommendation_id,
        ResponseApproval.decision == "pending"
    ).first()
    
    if not approval:
        raise HTTPException(status_code=400, detail="No pending approval request found")
        
    # Optional separation of duties check could go here
    # if approval.requested_by == current_user.username:
    #     raise HTTPException(status_code=403, detail="Requester cannot approve their own request")
        
    approval.decision = action.decision
    approval.reviewer = current_user.username
    approval.decision_at = datetime.now(UTC)
    approval.reason = action.reason
    
    if action.decision == "approved":
        recommendation.status = "approved"
        # Set expiration to 60 minutes from now
        approval.expiration = datetime.now(UTC) + timedelta(minutes=60)
    else:
        recommendation.status = "rejected"
        
    db.commit()
    db.refresh(approval)
    return approval

@router.post("/responses/{recommendation_id}/execute", response_model=ResponseExecutionResponse)
def execute_recommendation(
    recommendation_id: UUID4,
    execution_in: ResponseExecutionCreate,
    db: Session = Depends(dependencies.get_db),
    current_user: Principal = Depends(require_permission("responses:execute"))
) -> Any:
    """
    Execute a response (dry-run, simulation, analyst_approved_execution).
    """
    recommendation = db.query(ResponseRecommendation).filter(ResponseRecommendation.id == recommendation_id).first()
    if not recommendation:
        raise HTTPException(status_code=404, detail="Recommendation not found")
        
    if execution_in.mode == "analyst_approved_execution":
        if recommendation.status != "approved":
            raise HTTPException(status_code=400, detail="Execution requires an approved recommendation")
            
        approval = db.query(ResponseApproval).filter(
            ResponseApproval.recommendation_id == recommendation_id,
            ResponseApproval.decision == "approved"
        ).first()
        
        if not approval or (approval.expiration and approval.expiration < datetime.now(UTC)):
            raise HTTPException(status_code=400, detail="Valid unexpired approval required for execution")
    
    # Check for idempotency / avoid duplicate executions
    # In a full impl, idempotency key might be sent by the client, here we generate one or use standard pattern
    idempotency_key = f"{recommendation_id}_{execution_in.mode}"
    
    existing = db.query(ResponseExecution).filter(ResponseExecution.idempotency_key == idempotency_key).first()
    if existing:
        return existing
        
    execution = ResponseExecution(
        recommendation_id=recommendation.id,
        approval_id=approval.id if execution_in.mode == "analyst_approved_execution" else None,
        playbook_id=recommendation.playbook_id,
        action_type=recommendation.action_type,
        mode=execution_in.mode,
        target=recommendation.target_reference,
        executor="pending", # To be filled by worker
        idempotency_key=idempotency_key
    )
    db.add(execution)
    db.commit()
    db.refresh(execution)
    
    # NOTE: Here we would dispatch an event to `security.response.tasks` via Redpanda for the worker to pick up
    # For MVP we can just enqueue it in DB, and let worker poll, or send Kafka event.
    return execution

@router.post("/response-executions/{execution_id}/rollback")
def rollback_execution(
    execution_id: UUID4,
    db: Session = Depends(dependencies.get_db),
    current_user: Principal = Depends(require_permission("responses:execute"))
) -> Any:
    """
    Request rollback of a completed execution.
    """
    execution = db.query(ResponseExecution).filter(ResponseExecution.id == execution_id).first()
    if not execution:
        raise HTTPException(status_code=404, detail="Execution not found")
        
    if execution.status not in ["succeeded", "failed"]:
        raise HTTPException(status_code=400, detail="Cannot rollback execution in current state")
        
    execution.status = "rollback_pending"
    db.commit()
    # Event dispatched to worker here
    return {"message": "Rollback initiated"}
